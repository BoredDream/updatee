## C.2 问题一：确定性线性规划

问题一以计划购电费用最小为目标，并加入极小的储能吞吐惩罚，以减少线性规划退化时出现无意义的同时充放电。约束包括电量平衡、储能状态递推、充放电功率限制、SOC 上下界及日初日末 SOC 相等。

代码来源：[q1_solver.py](../../src/src/q1_solver.py#L90-L149)。

```python
n = cfg.intervals_per_day
price, net = inputs["price"], inputs["net"]

limits = [
    ("g", 0.0, None),
    ("c", 0.0, cfg.interval_limit_kwh),
    ("d", 0.0, cfg.interval_limit_kwh),
    ("u", 0.0, None),
]
base = {
    name: index * n
    for index, (name, _, _) in enumerate(limits)
}
soc0 = 4 * n
variable_count = 4 * n + n + 1

# 目标函数：计划购电费用与极小吞吐惩罚
objective = np.zeros(variable_count)
objective[base["g"]:base["g"] + n] = price
objective[base["c"]:base["c"] + n] = cfg.throughput_penalty
objective[base["d"]:base["d"] + n] = cfg.throughput_penalty

rows, cols, values, rhs = [], [], [], []
row = 0

# 供需平衡：g + d - c - u = 负载 - 光伏
for t in range(n):
    for name, coefficient in (
        ("g", 1.0),
        ("d", 1.0),
        ("c", -1.0),
        ("u", -1.0),
    ):
        rows.append(row)
        cols.append(base[name] + t)
        values.append(coefficient)
    rhs.append(float(net[t]))
    row += 1

# SOC递推：S[t+1] = S[t] + eta_c*c[t] - d[t]/eta_d
for t in range(n):
    rows.append(row); cols.append(soc0 + t + 1); values.append(1.0)
    rows.append(row); cols.append(soc0 + t); values.append(-1.0)
    rows.append(row); cols.append(base["c"] + t); values.append(-cfg.eta_charge)
    rows.append(row); cols.append(base["d"] + t); values.append(1.0 / cfg.eta_discharge)
    rhs.append(0.0)
    row += 1

# 日初、日末SOC相等
rows += [row, row]
cols += [soc0, soc0 + n]
values += [1.0, -1.0]
rhs.append(0.0)
row += 1

matrix = coo_matrix(
    (values, (rows, cols)),
    shape=(row, variable_count),
).tocsr()

bounds = []
for _, low, high in limits:
    bounds += [(low, high)] * n
bounds += (
    [(initial_soc, initial_soc)]
    + [(cfg.soc_min_kwh, cfg.soc_max_kwh)] * (n - 1)
    + [(initial_soc, initial_soc)]
)

result = linprog(
    objective,
    A_eq=matrix,
    b_eq=np.asarray(rhs),
    bounds=bounds,
    method="highs",
)
if not result.success:
    raise RuntimeError(f"第一问线性规划失败：{result.message}")
```

## C.3 问题二：历史残差情景与因果补救

问题二先建立 145 时段的中心预测，再将过去完整历史日的残差路径整体叠加到中心预测上。整条残差路径被保留，而不是对各时段独立抽样，从而保留预测误差的时序相关性。为防止未来信息泄漏，0:00 制定计划时最晚只使用 `day-2` 的完整历史路径。

代码来源：[q2_solver.py：情景构造](../../src/src/q2_solver.py#L162-L170)。

```python
def scenario_set(net, cold, day, weight, cfg):
    center = np.r_[
        point_forecast(net, cold, day, day, weight),
        point_forecast(net, cold, day + 1, day, weight)[0],
    ]

    # day-1路径包含当前尚未实现的午夜段，最晚只使用day-2
    residuals = {
        j: delivery_residual(net, cold, j, weight)
        for j in range(1, max(1, day - 1))
    }
    window = choose_window(residuals, day, cfg)
    indices = [
        j
        for j in range(max(1, day - 1 - window), day - 1)
        if j in residuals
    ]

    if not indices:
        return center[None, :], center, window

    scenarios = center[None, :] + np.stack(
        [residuals[j] for j in indices]
    )
    return scenarios, center, window
```

实际净负荷到达后，执行层按照“先取消不必要的充放电，再利用储能，最后紧急购电或弃电”的顺序进行因果补救。

代码来源：[q2_solver.py：因果调度](../../src/src/q2_solver.py#L229-L249)。

```python
def causal_dispatch(actual, grid, ref_charge, ref_discharge,
                    initial_soc, cfg):
    count = len(actual)
    charge = np.zeros(count)
    discharge = np.zeros(count)
    emergency = np.zeros(count)
    surplus = np.zeros(count)
    soc = np.zeros(count + 1)
    soc[0] = initial_soc

    for t in range(count):
        c = max(float(ref_charge[t]), 0.0)
        d = max(float(ref_discharge[t]), 0.0)

        common = min(c, d)
        c -= common
        d -= common
        c = min(
            c,
            cfg.interval_limit_kwh,
            (cfg.soc_max_kwh - soc[t]) / cfg.eta_charge,
        )
        d = min(
            d,
            cfg.interval_limit_kwh,
            (soc[t] - cfg.soc_min_kwh) * cfg.eta_discharge,
        )

        gap = float(actual[t] - (grid[t] + d - c))
        if gap > 0:
            # 缺电：先取消充电，再增加放电，最后紧急购电
            reduction = min(c, gap)
            c -= reduction
            gap -= reduction
            extra = max(0.0, min(
                gap,
                cfg.interval_limit_kwh - d,
                (soc[t] - cfg.soc_min_kwh)
                * cfg.eta_discharge - d,
            ))
            d += extra
            gap -= extra
            emergency[t] = max(gap, 0.0)
        elif gap < 0:
            # 富余：先取消放电，再增加充电，最后弃电
            remainder = -gap
            reduction = min(d, remainder)
            d -= reduction
            remainder -= reduction
            extra = max(0.0, min(
                remainder,
                cfg.interval_limit_kwh - c,
                (cfg.soc_max_kwh - soc[t])
                / cfg.eta_charge - c,
            ))
            c += extra
            remainder -= extra
            surplus[t] = max(remainder, 0.0)

        charge[t], discharge[t] = c, d
        soc[t + 1] = (
            soc[t]
            + cfg.eta_charge * c
            - d / cfg.eta_discharge
        )

    return {
        "charge": charge,
        "discharge": discharge,
        "emergency": emergency,
        "surplus": surplus,
        "soc": soc,
    }
```

## C.4 问题三：四阶段滚动随机规划

问题三在 0:00、6:00、12:00 和 18:00 更新计划。每次更新只锁定到下一次实际启用的决策时刻，之后的合同量作为情景追索变量。该实现属于逐阶段重新求解的滚动两阶段随机规划，而不是预先构造完整情景树的严格多阶段随机规划。

代码来源：[q3_multistage.py](../../src/src/q3_multistage.py#L463-L487)。

```python
TSTAGE = [0, 35, 71, 107, 144]

def solve_day(d, S0, committed_q, K=30,
              stages=(0, 1, 2, 3), cache=None):
    enabled = tuple(sorted(set(int(m) for m in stages)))
    if (
        not enabled
        or enabled[0] != 0
        or any(m not in (0, 1, 2, 3) for m in enabled)
    ):
        raise ValueError("stages必须包含0，且只能取0/1/2/3")

    first_adjust = (
        TSTAGE[enabled[1]] if len(enabled) > 1 else T
    )
    out = {
        k: np.zeros(T)
        for k in ("z", "c", "g", "w", "S")
    }

    # 0:00制定初始计划
    sl, sg = scenarios0_145(d, K)
    x = stage0_lp_145(
        committed_q,
        S0,
        sl,
        sg,
        commit_end=first_adjust,
    )
    q = x.copy()

    # 执行上一自然日已经锁定的午夜区间
    ml, mg = midnight_actual(d)
    S, midnight = dispatch_locked_interval(
        committed_q, S0, ml, mg
    )
    S_after_midnight = S

    # 在启用的日内阶段重新预测、构造情景并更新合同
    for m in range(4):
        tm, tn = TSTAGE[m], TSTAGE[m + 1]
        if m >= 1 and m in enabled:
            later = [j for j in enabled if j > m]
            commit_end = (
                TSTAGE[later[0]] if later else T
            )
            sl, sg = scenarios(d, m, K)
            q[tm:commit_end] = stage_lp(
                m,
                x,
                S,
                sl,
                sg,
                commit_end=commit_end,
            )

        # 实际值只在对应区间到达后用于执行
        S = dispatch(
            tm,
            min(tn, T - 1),
            q,
            S,
            L[d],
            G[d],
            out,
        )

    natural = {
        k: np.r_[midnight[k], out[k][:T - 1]]
        for k in ("c", "g", "z", "w")
    }
    natural["S"] = np.r_[
        S0, S_after_midnight, out["S"][:T - 1]
    ]
    return x, q, out, natural, S
```

def price_forecast_145(data, day: int, window: int = 20):
    p4 = data["price4_template"]
    p1 = data["price_template"]

    # day-1行的末列属于当前0:00尚未实现的区间
    end = day - 1
    if end <= 0:
        template = p1.copy()
    else:
        hist = p4[max(0, end - window):end]
        shape = hist.mean(axis=0) / max(
            float(hist.mean()), 1e-9
        )
        level = (
            0.5 * float(p4[end - 1].mean())
            + 0.5 * float(hist.mean())
        )
        template = np.maximum(shape * level, 1e-3)

    midnight = (
        float(p4[day - 1, 142])
        if day > 0
        else float(p1[-1])
    )
    return np.r_[midnight, template]

## C.6 问题 4-3：联合场景与方案 A 价格目标

问题 4-3 将负载、光伏和电价使用同一个历史日进行配对，形成联合情景。电价采用乘性残差，以保证预测价格为正。

代码来源：[q4_solver.py：联合场景](../../src/src/q4_solver.py#L152-L167)。

```python
def scenarios4(d, m, K=30):
    Lh = load_hat(d, m)
    Gh = pv_hat(d, m)
    Ph = price_hat(d, m)
    sl, sg, spz = [], [], []
    first_lag = 2 if m == 0 else 1

    for j in range(first_lag, first_lag + K):
        dd = d - j
        if dd < 10:
            continue

        sl.append(np.maximum(
            Lh + L[dd] - load_hat(dd, m), 0.0
        ))
        sg.append(np.maximum(
            Gh + G[dd] - pv_hat(dd, m), 0.0
        ))

        # 乘性价格残差，并限制极端比例
        ratio = PMAT[dd] / np.maximum(
            price_hat(dd, m), 1e-6
        )
        spz.append(np.maximum(
            Ph * np.clip(ratio, 0.3, 3.0), 1e-3
        ))

    if not sl:
        sl, sg, spz = [Lh], [Gh], [Ph]
    return sl, sg, spz
```
